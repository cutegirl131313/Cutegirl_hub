	end
end

local function enableDrag(btn)
	local drag, from, base
	btn.InputBegan:Connect(function(input)
		if locked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			drag = true
			from = input.Position
			base = btn.Position
		end
	end)
	btn.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			drag = false
		end
	end)
	UIS.InputChanged:Connect(function(input)
		if not drag or locked then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			local d = input.Position - from
			btn.Position = UDim2.fromOffset(base.X.Offset + d.X, base.Y.Offset + d.Y)
		end
	end)
end
enableDrag(toggleBtn)
enableDrag(lockBtn)

local pages, tabBtns, cur = {}, {}, nil

local function pick(name)
	cur = name
	for n, p in pairs(pages) do p.Visible = (n == name) end

		b.TextColor3 = (n == name) and C.accent or C.dim
	end
end

local function addToggle(parent, key, label, default)
	S[key] = default and true or false
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, 0, 0, 24)
	row.BackgroundTransparency = 1
	row.Parent = parent
	local lab = Instance.new("TextLabel")

	lab.BackgroundTransparency = 1
	lab.Text = label
	lab.Font = Enum.Font.Gotham
	lab.TextSize = 12
	lab.TextColor3 = C.txt
	lab.TextXAlignment = Enum.TextXAlignment.Left
	lab.Parent = row
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.fromOffset(40, 20)
	btn.Position = UDim2.new(1, -40, 0.5, -10)
	-- ON = yellow, OFF = gray (always, even after config load)
	local ON_COL = Color3.fromRGB(255, 200, 40)
	local OFF_COL = Color3.fromRGB(70, 70, 78)
	btn.BackgroundColor3 = S[key] and ON_COL or OFF_COL
	btn.BorderSizePixel = 0
	btn.Text = S[key] and "ON" or "OFF"
	btn.Font = Enum.Font.GothamBold
	btn.TextSize = 10
	btn.TextColor3 = S[key] and Color3.fromRGB(20, 20, 20) or Color3.fromRGB(160, 160, 165)
	btn.Parent = row
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, 0)
	c.Parent = btn
	local function set(v)
		S[key] = v and true or false
		btn.Text = S[key] and "ON" or "OFF"
		btn.BackgroundColor3 = S[key] and ON_COL or OFF_COL
		btn.TextColor3 = S[key] and Color3.fromRGB(20, 20, 20) or Color3.fromRGB(160, 160, 165)
	end
	btn.MouseButton1Click:Connect(function() set(not S[key]) end)
	UIRefs[key] = set
	return set
end

local function addSlider(parent, key, label, default, minv, maxv)
	S[key] = default
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, 0, 0, 36)
	row.BackgroundTransparency = 1
	row.Parent = parent
	local lab = Instance.new("TextLabel")
	lab.Size = UDim2.new(1, 0, 0, 14)
	lab.BackgroundTransparency = 1
	lab.Text = label .. ": " .. tostring(default)
	lab.Font = Enum.Font.Gotham
	lab.TextSize = 11
	lab.TextColor3 = C.txt
	lab.TextXAlignment = Enum.TextXAlignment.Left
	lab.Parent = row
	local bar = Instance.new("Frame")
	bar.Size = UDim2.new(1, 0, 0, 8)
	bar.Position = UDim2.fromOffset(0, 20)
	bar.BackgroundColor3 = Color3.fromRGB(50, 50, 58)
	bar.BorderSizePixel = 0
	bar.Parent = row
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, 0)
	c.Parent = bar
	local fill = Instance.new("Frame")
	fill.Size = UDim2.new(math.clamp((default - minv) / math.max(maxv - minv, 1), 0, 1), 0, 1, 0)
	fill.BackgroundColor3 = C.accent
	fill.BorderSizePixel = 0
	fill.Parent = bar
	local c2 = Instance.new("UICorner")
	c2.CornerRadius = UDim.new(0, 0)
	c2.Parent = fill
	local function set(v)
		v = math.clamp(tonumber(v) or minv, minv, maxv)
		S[key] = math.floor(v + 0.5)
		fill.Size = UDim2.new((S[key] - minv) / math.max(maxv - minv, 1), 0, 1, 0)
		lab.Text = label .. ": " .. tostring(S[key])
	end
	bar.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
		local conn, endc
		local function upd(pos)
			local rel = math.clamp((pos.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
			set(minv + rel * (maxv - minv))
		end
		upd(input.Position)
		conn = UIS.InputChanged:Connect(function(inp)
			if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
				upd(inp.Position)
			end
		end)
		endc = UIS.InputEnded:Connect(function(inp)
			if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
				if conn then conn:Disconnect() end
				if endc then endc:Disconnect() end
			end
		end)
	end)
	UIRefs[key] = set
	return set
end

local function ensureDir()
	pcall(function()
		if typeof(makefolder) == "function" then
			makefolder("VK_CuteGirl")

		end
	end)

local function cfgPath(name)
	name = tostring(name or "default"):gsub("[^%w%-%_]", "")
	if name == "" then name = "default" end
	return CONFIG_DIR .. "/" .. name .. ".json"
end

local function listConfigs()
	local names = {}
	pcall(function()
		if typeof(listfiles) == "function" then
			ensureDir()
			for _, f in ipairs(listfiles(CONFIG_DIR)) do
				local n = string.match(f, "([^/\\]+)%.json$")
				if n then table.insert(names, n) end
			end
		end
	end)
	if #names == 0 and _G.VK_CuteGirl_Configs then
		for n in pairs(_G.VK_CuteGirl_Configs) do table.insert(names, n) end
	end
	table.sort(names)
	return names
end

local refreshCfgList -- fwd

local function saveConfig(name)
	name = name or selectedConfig or "default"
	name = tostring(name):gsub("[^%w%-%_]", "")
	if name == "" then name = "default" end
	selectedConfig = name
	local data = {}
	for k, v in pairs(S) do
		if type(v) == "boolean" or type(v) == "number" or type(v) == "string" then
			data[k] = v
		end
	end
	data._theme = currentTheme or "original"
	local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
	if not ok then encoded = "{}" end
	ensureDir()
	local wrote = false
	pcall(function()
		if typeof(writefile) == "function" then
			writefile(cfgPath(name), encoded)
			pcall(function() writefile(CONFIG_LAST, name) end)
			wrote = true
		end
	end)
	if not wrote then
		_G.VK_CuteGirl_Configs = _G.VK_CuteGirl_Configs or {}
		_G.VK_CuteGirl_Configs[name] = encoded
		_G.VK_CuteGirl_Last = name
		wrote = true
	end
	if refreshCfgList then refreshCfgList() end
	notify(wrote and ("saved: " .. name) or "save failed")
end

local function applyConfigData(data)
	if type(data) ~= "table" then return end
	for k, v in pairs(data) do
		if k == "_theme" then
			if Themes[v] then applyTheme(v) end
		elseif S[k] ~= nil and UIRefs[k] then
			UIRefs[k](v)
		elseif S[k] ~= nil then
			S[k] = v
		end
	end
end

local function loadConfig(name)
	name = name or selectedConfig or "default"
	selectedConfig = name
	local raw
	pcall(function()
		if typeof(readfile) == "function" and typeof(isfile) == "function" and isfile(cfgPath(name)) then
			raw = readfile(cfgPath(name))
		end
	end)
	if not raw and _G.VK_CuteGirl_Configs and _G.VK_CuteGirl_Configs[name] then
		raw = _G.VK_CuteGirl_Configs[name]
	end
	if not raw then notify("no config: " .. tostring(name)) return end
	local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
	if ok and type(data) == "table" then
		applyConfigData(data)
		pcall(function() if typeof(writefile) == "function" then writefile(CONFIG_LAST, name) end end)
		_G.VK_CuteGirl_Last = name
		notify("loaded: " .. name)
	else
		notify("bad config")
	end
end

local function deleteConfig(name)
	name = name or selectedConfig or "default"
	local deleted = false
	pcall(function()
		if typeof(delfile) == "function" and typeof(isfile) == "function" and isfile(cfgPath(name)) then
			delfile(cfgPath(name))
			deleted = true
		end
	end)
	if _G.VK_CuteGirl_Configs then _G.VK_CuteGirl_Configs[name] = nil end
	if refreshCfgList then refreshCfgList() end
	notify(deleted and ("deleted: " .. name) or "cleared: " .. name)
end

local function tryAutoload()
	-- only if user selected a config name (not "off")
	local last = S.AutoLoadName
	if (not last or last == "" or last == "off") then
		pcall(function()
			if typeof(readfile) == "function" and typeof(isfile) == "function" and isfile(CONFIG_LAST) then
				last = readfile(CONFIG_LAST)
			end
		end)
	end
	if (not last or last == "" or last == "off") and _G.VK_CuteGirl_Last then
		last = _G.VK_CuteGirl_Last
	end

	pcall(function() nameBox.Text = last end)
	pcall(function() if autoPickBtn then autoPickBtn.Text = last end end)
	loadConfig(last)
	pcall(refreshCfgList)
end

local function getRootPart()
	local lf = getFighter()
	if lf and lf.Entity and lf.Entity.RootPart then return lf.Entity.RootPart end
	return LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
end

local function validEnemy(plr)
	if plr == LP then return false end
	local char = plr.Character
	if not char then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum or not hrp or hum.Health <= 0 then return false end
	if S.TeamCheck then
		local a, b = LP:GetAttribute("TeamID"), plr:GetAttribute("TeamID")
		if a ~= nil and b ~= nil and a == b then return false end
		local ok, same = pcall(function()
			return LP.Team and plr.Team and LP.Team == plr.Team
		end)
		if ok and same then return false end
	end
	return true
end

local function hasLOS(fromPos, toPos, targetChar)
	if not S.WallCheck then return true end
	if typeof(fromPos) ~= "Vector3" or typeof(toPos) ~= "Vector3" then return false end
	local dir = toPos - fromPos
	local dist = dir.Magnitude
	if dist < 1 then return true end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local filter = {}
	if LP.Character then table.insert(filter, LP.Character) end
	if targetChar then table.insert(filter, targetChar) end
	params.FilterDescendantsInstances = filter
	params.IgnoreWater = true
	return Workspace:Raycast(fromPos, dir.Unit * dist, params) == nil
end

local function getHead(plr)
	local c = plr.Character
	if not c then return nil end
	return c:FindFirstChild("HitboxHead")
		or c:FindFirstChild("HitboxHeadSmall")
		or c:FindFirstChild("Head")
end

local function getNearest(ignoreWall)
	local root = getRootPart()
	local cam = Workspace.CurrentCamera
	local from = (root and root.Position) or (cam and cam.CFrame.Position)
	if not from then return nil end
	local best, bd = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if validEnemy(plr) then
			local head = getHead(plr)
			local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
			if head and hrp then
				if ignoreWall or hasLOS(from, head.Position, plr.Character) then
					local d = (from - hrp.Position).Magnitude
					if d < bd then bd, best = d, head end
				end
			end
		end
	end
	return best
end

local function getCrosshair(fov)
	local cam = Workspace.CurrentCamera
	if not cam then return nil end
	local best, bd = nil, (fov or 300)
	local center = cam.ViewportSize / 2
	local from = cam.CFrame.Position
	for _, plr in ipairs(Players:GetPlayers()) do
		if validEnemy(plr) then
			local head = getHead(plr)
			if head then
				local sp, on = cam:WorldToViewportPoint(head.Position)
				if on and sp.Z > 0 then
					local dist = (Vector2.new(sp.X, sp.Y) - Vector2.new(center.X, center.Y)).Magnitude
					if dist <= bd then
						if (not S.WallCheck) or hasLOS(from, head.Position, plr.Character) then
							bd, best = dist, head
						end
					end
				end
			end
		end
	end
	return best
end

-- ========== Silent (Solar-style, fast + dagger) ==========
local function solarClosestHead()
	-- Solar get_closest: on-screen nearest to crosshair center
	local cam = Workspace.CurrentCamera
	if not cam then return nil end
	local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
	local fov = tonumber(S.AimFOV) or 400
	pcall(function()
		if cam.FieldOfView and cam.FieldOfView <= 60 then
			fov = math.max(fov * 2, 700)
		end
	end)
	local best, bestD = nil, fov
	local function consider(model)
		if not model or model == LP.Character then return end
		local hum = model:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then return end
		local hrp = model:FindFirstChild("HumanoidRootPart")
		if not hrp then return end
		-- team check via player
		local plr = Players:GetPlayerFromCharacter(model)
		if plr then
			if S.TeamCheck then
				local a, b = LP:GetAttribute("TeamID"), plr:GetAttribute("TeamID")
				if a and b and a == b then return end
				if LP.Team and plr.Team and LP.Team == plr.Team then return end
			end
		end
		local head = model:FindFirstChild("HitboxHead")
			or model:FindFirstChild("HitboxHeadSmall")
			or model:FindFirstChild("Head")
		if not head then return end
		local sp, on = cam:WorldToViewportPoint(head.Position)
		if not on or sp.Z <= 0 then return end
		local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
		if d < bestD then
			-- wall check only if enabled
			if S.WallCheck then
				local params = RaycastParams.new()
				params.FilterType = Enum.RaycastFilterType.Exclude
				params.FilterDescendantsInstances = { LP.Character, model }
				local hit = Workspace:Raycast(cam.CFrame.Position, head.Position - cam.CFrame.Position, params)
				if hit then return end
			end
			bestD = d
			best = head
		end
	end
	for _, child in ipairs(Workspace:GetChildren()) do
		if child:FindFirstChildOfClass("Humanoid") then
			consider(child)
		elseif child.Name == "HurtEffect" then
			for _, hp in ipairs(child:GetChildren()) do
				if hp.ClassName ~= "Highlight" then consider(hp) end
			end
		end
	end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character then consider(plr.Character) end
	end
	return best
end

local function pickSilentHead()
	return solarClosestHead()
end

local function pickRageHead()
	local best, bestD = nil, 1e9
	local root = getRootPart and getRootPart()
	local cam = Workspace.CurrentCamera
	local ref = (root and root.Position) or (cam and cam.CFrame.Position)
	if not ref then return nil end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP then
			local char = plr.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			if hum and hrp and hum.Health > 0 then
				local skip = false
				if S.TeamCheck then
					local a, b = LP:GetAttribute("TeamID"), plr:GetAttribute("TeamID")
					if a and b and a == b then skip = true end
					if LP.Team and plr.Team and LP.Team == plr.Team then skip = true end
				end
				if not skip then
					local d = (ref - hrp.Position).Magnitude
					if d < bestD and d < 5000 then
						bestD = d
						best = char:FindFirstChild("HitboxHead")
							or char:FindFirstChild("HitboxHeadSmall")
							or char:FindFirstChild("Head")
							or hrp
					end
				end
			end
		end
	end
	return best
end

local function encodeShot(from, to, head)
	if typeof(from) ~= "Vector3" or typeof(to) ~= "Vector3" or not head then return nil end
	refreshMods()
	if Utility and type(Utility.EncodeCFrame) == "function" then
		local ok, data = pcall(function()
			local look = CFrame.lookAt(from, to)
			local e = Utility:EncodeCFrame(look)
			local e2 = Utility:EncodeCFrame(CFrame.new(to))
			local rel = head.CFrame:ToObjectSpace(CFrame.new(to))
			local e3 = Utility:EncodeCFrame(rel)
			return {
				[utf8.char(1)] = {
					[utf8.char(0)] = e,
					[utf8.char(1)] = e2,
					[utf8.char(2)] = head,
					[utf8.char(3)] = e3,
				},
			}
		end)
		if ok and data then return data end
	end
	return nil
end

local function fireAt(head, mode)
	if not head or not head.Parent then return false end
	refreshMods()
	if not UseItem or not EnumLibrary then return false end
	local okE, se = pcall(function() return EnumLibrary:ToEnum("StartShooting") end)
	if not okE or se == nil then return false end
	local lf = getFighter and getFighter()
	if not lf or not lf.EquippedItem then return false end
	local oid
	pcall(function()
		if lf.EquippedItem.Get then oid = lf.EquippedItem:Get("ObjectID") end
		if not oid and lf.EquippedItem.Data then oid = lf.EquippedItem.Data.ObjectID end
	end)
	if not oid then return false end
	local cam = Workspace.CurrentCamera
	local root = getRootPart and getRootPart()
	local targetPos = head.Position
	local from = (cam and cam.CFrame.Position) or (root and root.Position) or targetPos
	if mode == "target" then
		from = targetPos + Vector3.new(0, 0.15, 0)
	end
	local data = encodeShot(from, targetPos, head)
	if not data then return false end
	return pcall(function() UseItem:FireServer(oid, se, data, nil) end)
end

local function fireProjectileAt(head)
	if not head or not head.Parent then return false end
	refreshMods()
	if not (Utility and EnumLibrary and UseItem and Utility.EncodeCFrame) then return false end
	local okE, se = pcall(function() return EnumLibrary:ToEnum("StartShooting") end)
	if not okE or se == nil then return false end
	local lf = getFighter and getFighter()
	if not lf or not lf.EquippedItem then return false end
	local oid
	pcall(function()
		if lf.EquippedItem.Get then oid = lf.EquippedItem:Get("ObjectID") end
		if not oid and lf.EquippedItem.Data then oid = lf.EquippedItem.Data.ObjectID end
	end)
	if not oid then return false end
	local model = head:FindFirstAncestorOfClass("Model") or head.Parent
	local hrp = model and model:FindFirstChild("HumanoidRootPart")
	local pos = head.Position
	-- lead for slow projectiles
	pcall(function()
		if hrp then
			pos = pos + hrp.AssemblyLinearVelocity * ((tonumber(S.Prediction) or 20) / 100)
		end
	end)
	local origin = pos - Vector3.new(0, 3.5, 0)
	local ok = false
	-- Solar arcade flat packet
	pcall(function()
		local packet = {
			[utf8.char(0)] = Utility:EncodeCFrame(CFrame.new(origin, pos)),
			[utf8.char(1)] = Utility:EncodeCFrame(CFrame.new(pos)),
			[utf8.char(2)] = head,
			[utf8.char(3)] = Utility:EncodeCFrame(head.CFrame:ToObjectSpace(CFrame.new(pos))),
		}
		if UseItem:FireServer(oid, se, packet, nil) then ok = true end
	end)
	-- nested packet too
	pcall(function()

		if data then
			UseItem:FireServer(oid, se, data, nil)
			ok = true
		end
	end)
	-- call module StartShooting directly
	pcall(function()
		local ps = LP:FindFirstChild("PlayerScripts")
		local it = ps and ps:FindFirstChild("Modules")
		it = it and it:FindFirstChild("ItemTypes")
		if not it then return end
		for _, name in ipairs({ "Slingshot", "Bow", "Daggers", "Gun", "Melee", "Projectile", "Throwable" }) do
			local m = it:FindFirstChild(name)
			if m then
				local okm, mod = pcall(require, m)
				if okm and mod then
					if mod.StartShooting then pcall(function() mod:StartShooting() end) end
					if mod.Fire then pcall(function() mod:Fire() end) end
					if mod.Shoot then pcall(function() mod:Shoot() end) end
					if mod.Throw then pcall(function() mod:Throw() end) end
				end
			end
		end
	end)
	return ok
end

RunService.RenderStepped:Connect(function(dt)
	if not S.Fly then
		if flyBV then stopFly() end
		return
	end
	startFly()
	local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
	local cam = Workspace.CurrentCamera
	if not hrp or not cam then return end

	local speed = tonumber(S.FlySpeed) or 60
	local look, right = cam.CFrame.LookVector, cam.CFrame.RightVector
	local move = Vector3.zero

	-- move keys follow camera (look up + W = go up, no separate up/down buttons)
	if UIS:IsKeyDown(Enum.KeyCode.W) then move = move + look end
	if UIS:IsKeyDown(Enum.KeyCode.S) then move = move - look end
	if UIS:IsKeyDown(Enum.KeyCode.A) then move = move - right end
	if UIS:IsKeyDown(Enum.KeyCode.D) then move = move + right end
	-- mobile: joystick direction projected onto camera axes (includes vertical from look)
	if hum and hum.MoveDirection.Magnitude > 0.05 then
		local md = hum.MoveDirection
		local flat = Vector3.new(md.X, 0, md.Z)
		if flat.Magnitude > 0.05 then
			local f = Vector3.new(look.X, 0, look.Z)
			local r = Vector3.new(right.X, 0, right.Z)
			if f.Magnitude > 0.05 then f = f.Unit else f = Vector3.new(0, 0, -1) end
			if r.Magnitude > 0.05 then r = r.Unit else r = Vector3.new(1, 0, 0) end
			-- forward amount along camera look (with Y), strafe along right
			local forwardAmt = flat.Unit:Dot(f)
			local strafeAmt = flat.Unit:Dot(r)
			move = move + look * forwardAmt + right * strafeAmt
		end
	end

	local vel = move.Magnitude > 0 and move.Unit * speed or Vector3.zero
	if flyBV and flyBV.Parent then
		flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		flyBV.Velocity = vel
	end
	if flyBG and flyBG.Parent then
		flyBG.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + look)
	end
	hrp.AssemblyLinearVelocity = vel
	if vel.Magnitude > 0 then
		hrp.CFrame = hrp.CFrame + vel * math.clamp(dt, 0, 0.05)
	end
	if hum then
		hum.PlatformStand = true
	end
end)

local function isEnemy(plr)
	if plr == LP then return false end
	local char = plr.Character
	if not char then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return false end
	if S.TeamCheck then
		local a, b = LP:GetAttribute("TeamID"), plr:GetAttribute("TeamID")
		if a ~= nil and b ~= nil and a == b then return false end
		if LP.Team and plr.Team and LP.Team == plr.Team then return false end
	end
	if S.ImmunityCheck then
		local skip = false
		pcall(function()
			if char:GetAttribute("Immunity") or char:GetAttribute("Invincible") then skip = true end
		end)
		if skip then return false end
	end
	return true
end

local function getEnemyWeapon(plr)
	local name = nil
	pcall(function()
		local char = plr.Character
		if not char then return end
		local tool = char:FindFirstChildOfClass("Tool")
		if tool then name = tool.Name return end
		for _, c in ipairs(char:GetChildren()) do
			if c:IsA("Model") or c:IsA("Folder") then
				local n = string.lower(c.Name)
				if n:find("gun") or n:find("knife") or n:find("sword") or n:find("rifle")
					or n:find("pistol") or n:find("sniper") or n:find("melee") or n:find("weapon") then
					name = c.Name
					return
				end
			end
		end
		local attr = char:GetAttribute("EquippedItem") or char:GetAttribute("Weapon") or char:GetAttribute("HeldItem")
		if attr then name = tostring(attr) end
	end)
	return name
end

						h = math.clamp(h, 24, 320)
						local w = h * 0.55
						local x = spT.X - w / 2
						local y = spT.Y
						e.box.Visible = true
						e.box.Position = UDim2.fromOffset(math.floor(x), math.floor(y))
						e.box.Size = UDim2.fromOffset(math.floor(w), math.floor(h))

						e.box.BackgroundTransparency = 1
						if e.boxOut then e.boxOut.Visible = showBox end
						if e.boxIn then e.boxIn.Visible = showBox end
						if e.stroke then
							e.stroke.Enabled = showBox
							e.stroke.Color = col
							e.stroke.Thickness = 0.8
						end
						if e.strokeOut then
							e.strokeOut.Enabled = showBox
							e.strokeOut.Color = Color3.fromRGB(0, 0, 0)

						end
						if e.hl then
							e.hl.Enabled = false -- pure box style
						end

						if e.name then
							e.name.Visible = (S.ESP_Name ~= false)
							e.name.Text = string.upper(string.sub(plr.DisplayName or plr.Name, 1, 14))
						end

						-- vertical HP

							local show = (S.ESP_HealthBar ~= false)

								local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
								e.hpFill.Size = UDim2.new(1, 0, pct, 0)

									e.hpFill.BackgroundColor3 = Color3.fromRGB(230, 180, 50)
								else
									e.hpFill.BackgroundColor3 = Color3.fromRGB(230, 60, 60)
								end
							end
						end

							e.dist.Visible = (S.ESP_Distance ~= false)
							if myRoot then
								e.dist.Text = tostring(math.floor((myRoot.Position - hrp.Position).Magnitude)) .. "m"

local lastHP = {}
local function watchHumanoid(plr, hum)
	if not hum or hum:GetAttribute("VK_HitWatch") then return end
	hum:SetAttribute("VK_HitWatch", true)
	local id = plr.UserId
	lastHP[id] = hum.Health
	hum.HealthChanged:Connect(function(hp)
		if not (S.HitNotify or S.HitSound or S.KillSound) then
			lastHP[id] = hp
			return
		end
		local prev = lastHP[id]
		if prev and hp < prev - 0.2 then
			local dmg = math.floor(prev - hp + 0.5)
			if S.HitNotify then
				showHitNotify(string.format("HIT  %s  -%d", plr.DisplayName or plr.Name, dmg), Color3.fromRGB(255, 80, 50))
			end
			if S.HitSound then
				playPack(HIT_SOUNDS, S.HitSoundPack or "click")
			end
		end
		if prev and prev > 0 and hp <= 0 then
			if S.HitNotify then
				showHitNotify("KILL  " .. (plr.DisplayName or plr.Name), Color3.fromRGB(255, 40, 40))
			end
			if S.KillSound then
				playPack(KILL_SOUNDS, S.KillSoundPack or "deep")
			end
		end
		lastHP[id] = hp
	end)
	hum.Died:Connect(function()
		if not (S.HitNotify or S.KillSound) then return end
		local prev = lastHP[id]
		if prev and prev > 0 then
			if S.HitNotify then
				showHitNotify("KILL  " .. (plr.DisplayName or plr.Name), Color3.fromRGB(255, 40, 40))
			end
			if S.KillSound then
				playPack(KILL_SOUNDS, S.KillSoundPack or "deep")
			end
		end
		lastHP[id] = 0
	end)
end

local function hookPlayer(plr)
	if plr == LP then return end
	local function onChar(char)
		task.defer(function()
			local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 5)
			if hum then watchHumanoid(plr, hum) end
		end)
	end
	if plr.Character then onChar(plr.Character) end
	plr.CharacterAdded:Connect(onChar)
end
for _, plr in ipairs(Players:GetPlayers()) do hookPlayer(plr) end
Players.PlayerAdded:Connect(hookPlayer)

		if plr ~= LP and plr.Character then

						if S.HitNotify then
							showHitNotify("KILL  " .. (plr.DisplayName or plr.Name), Color3.fromRGB(255, 40, 40))
						end
					end
				end

local lastSky, savedLighting = nil, nil
local function applySkybox(name)
	if name == lastSky then return end
	lastSky = name
	pcall(function()
		-- remove ALL existing skies so game sky doesn't cover ours
		for _, c in ipairs(Lighting:GetChildren()) do

local lightingBackup = nil
local function saveLighting()
	if lightingBackup then return end
	pcall(function()
		lightingBackup = {
			Ambient = Lighting.Ambient,
			OutdoorAmbient = Lighting.OutdoorAmbient,
			Brightness = Lighting.Brightness,
			ClockTime = Lighting.ClockTime,
			FogEnd = Lighting.FogEnd,
			FogStart = Lighting.FogStart,
			GlobalShadows = Lighting.GlobalShadows,
		}
	end)
end
local function restoreLighting()
	if not lightingBackup then return end
	pcall(function()
		for k, v in pairs(lightingBackup) do
			Lighting[k] = v
		end
	end)
end
local lastAmbient, lastFB = nil, nil
local function applyAmbient(mode)
	if mode == lastAmbient and not S.Fullbright then return end
	lastAmbient = mode
	pcall(function()
		saveLighting()
		if mode == "dark" then
			Lighting.Ambient = Color3.fromRGB(30, 30, 40)
			Lighting.OutdoorAmbient = Color3.fromRGB(30, 30, 40)
			Lighting.Brightness = 0.5
		elseif mode == "bright" then
			Lighting.Ambient = Color3.fromRGB(180, 180, 190)
			Lighting.OutdoorAmbient = Color3.fromRGB(180, 180, 190)
			Lighting.Brightness = 3
		elseif mode == "red" then
			Lighting.Ambient = Color3.fromRGB(90, 25, 25)
			Lighting.OutdoorAmbient = Color3.fromRGB(90, 25, 25)
			Lighting.Brightness = 2
		elseif mode == "blue" then
			Lighting.Ambient = Color3.fromRGB(25, 45, 100)
			Lighting.OutdoorAmbient = Color3.fromRGB(25, 45, 100)
			Lighting.Brightness = 2
		else
			-- default: restore original
			restoreLighting()
		end
	end)
end

-- WalkSpeed / Jump: Rivals overwrites Humanoid often -> force every frame + velocity boost
local function applyMoveStats()
	local char = LP.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum then return end

	if S.WalkSpeed then
		local spd = math.clamp(tonumber(S.WalkSpeedVal) or 20, 1, 500)
		pcall(function()
			hum.WalkSpeed = spd
		end)
		-- velocity boost (works when game locks WalkSpeed)
		if hrp and hum.MoveDirection.Magnitude > 0.05 then
			local md = hum.MoveDirection
			local v = hrp.AssemblyLinearVelocity
			local target = md * spd
			hrp.AssemblyLinearVelocity = Vector3.new(target.X, v.Y, target.Z)
		end
	end

	if S.JumpPower then
		local jp = math.clamp(tonumber(S.JumpPowerVal) or 50, 1, 500)
		pcall(function()
			hum.UseJumpPower = true
			hum.JumpPower = jp
			hum.JumpHeight = math.clamp(jp / 50 * 7.2, 1, 80)
		end)
	end
end

-- highest priority so we win over game scripts
pcall(function()
	RunService:BindToRenderStep("VK_MoveStats", Enum.RenderPriority.Character.Value + 1, applyMoveStats)
end)
RunService.Heartbeat:Connect(applyMoveStats)
RunService.Stepped:Connect(function()
	applyMoveStats()
end)

-- re-apply when game tries to change WalkSpeed
local function hookHum(hum)
	if not hum or hum:GetAttribute("VK_MoveHook") then return end
	hum:SetAttribute("VK_MoveHook", true)
	hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
		if S.WalkSpeed then
			local spd = math.clamp(tonumber(S.WalkSpeedVal) or 20, 1, 500)
			if math.abs(hum.WalkSpeed - spd) > 0.5 then
				hum.WalkSpeed = spd
			end
		end
	end)
	hum:GetPropertyChangedSignal("JumpPower"):Connect(function()
		if S.JumpPower then
			local jp = math.clamp(tonumber(S.JumpPowerVal) or 50, 1, 500)
			if math.abs((hum.JumpPower or 0) - jp) > 0.5 then
				pcall(function() hum.JumpPower = jp end)
			end
		end
	end)
end
if LP.Character then
	local h = LP.Character:FindFirstChildOfClass("Humanoid")
	if h then hookHum(h) end
end
LP.CharacterAdded:Connect(function(char)
	task.defer(function()
		local h = char:WaitForChild("Humanoid", 5)
		if h then hookHum(h) end
		applyMoveStats()
	end)
end)

-- Infinite jump + jump power boost on jump
UIS.JumpRequest:Connect(function()
	local char = LP.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hum then return end
	if S.InfiniteJump then
		pcall(function() hum:ChangeState(Enum.HumanoidStateType.Jumping) end)
	end
	if S.JumpPower and hrp then
		local jp = math.clamp(tonumber(S.JumpPowerVal) or 50, 1, 500)
		local v = hrp.AssemblyLinearVelocity
		if v.Y < jp * 0.5 then
			hrp.AssemblyLinearVelocity = Vector3.new(v.X, jp * 0.85, v.Z)
		end
	end
end)

-- Third person: force camera behind character (Rivals locks FP)
local _thirdOn = false
local function setCharVisible(on)
	local char = LP.Character
	if not char then return end
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then
			pcall(function()
				d.LocalTransparencyModifier = on and 0 or d.LocalTransparencyModifier
			end)
		end
	end
end

RunService:BindToRenderStep("VK_ThirdPerson", Enum.RenderPriority.Camera.Value + 1, function()
	pcall(function()
		local cam = Workspace.CurrentCamera
		if not cam then return end
		if S.ThirdPerson and not S.Freecam then
			_thirdOn = true
			local char = LP.Character
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not hrp then return end
			setCharVisible(true)
			-- force zoom distances every frame
			LP.CameraMode = Enum.CameraMode.Classic
			LP.CameraMinZoomDistance = 12
			LP.CameraMaxZoomDistance = 16
			cam.CameraType = Enum.CameraType.Scriptable
			-- behind + up
			local look = cam.CFrame.LookVector
			-- keep mouse look: use current camera yaw if possible
			local dist = 12
			local height = 3
			local cf = hrp.CFrame
			-- rotate with character/move
			local back = -cf.LookVector * dist + Vector3.new(0, height, 0)
			local camPos = hrp.Position + back
			cam.CFrame = CFrame.new(camPos, hrp.Position + Vector3.new(0, 1.5, 0))
			if hum then cam.CameraSubject = hum end
		elseif _thirdOn then
			_thirdOn = false
			cam.CameraType = Enum.CameraType.Custom
			LP.CameraMinZoomDistance = 0.5
			LP.CameraMaxZoomDistance = 0.5
			task.defer(function()
				pcall(function()
					LP.CameraMaxZoomDistance = 128
					local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
					if hum then Workspace.CurrentCamera.CameraSubject = hum end
				end)
			end)
		end
	end)
end)

local freecamPart, freecamConn
local function stopFreecam()
	if freecamConn then pcall(function() freecamConn:Disconnect() end); freecamConn = nil end
	pcall(function()
		local cam = Workspace.CurrentCamera
		if cam and LP.Character then
			local hum = LP.Character:FindFirstChildOfClass("Humanoid")
			if hum then
				cam.CameraType = Enum.CameraType.Custom
				cam.CameraSubject = hum
			end
		end
		if freecamPart then freecamPart:Destroy(); freecamPart = nil end
	end)
end
local function startFreecam()
	stopFreecam()
	local cam = Workspace.CurrentCamera
	if not cam then return end
	freecamPart = Instance.new("Part")
	freecamPart.Anchored = true
	freecamPart.CanCollide = false
	freecamPart.Transparency = 1
	freecamPart.Size = Vector3.new(1,1,1)
	freecamPart.CFrame = cam.CFrame
	freecamPart.Parent = Workspace
	cam.CameraType = Enum.CameraType.Scriptable
	freecamConn = RunService.RenderStepped:Connect(function(dt)
		if not S.Freecam then stopFreecam() return end
		local speed = 60 * (dt or 0.016)
		local move = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then move = move + cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) then move = move - cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) then move = move - cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) then move = move + cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.E) then move = move + Vector3.new(0,1,0) end
		if UIS:IsKeyDown(Enum.KeyCode.Q) then move = move - Vector3.new(0,1,0) end
		if move.Magnitude > 0 then freecamPart.CFrame = freecamPart.CFrame + move.Unit * speed end
		cam.CFrame = freecamPart.CFrame
	end)
end
task.spawn(function()
	local last = false
	while task.wait(0.15) do
		if S.Freecam and not last then startFreecam() end
		if (not S.Freecam) and last then stopFreecam() end
		last = S.Freecam and true or false
	end
end)

RunService.RenderStepped:Connect(function()
	if not S.Tracers then
		for k,l in pairs(tracerLines) do pcall(function() l:Destroy() end); tracerLines[k]=nil end
		return
	end
	local cam = Workspace.CurrentCamera
	if not cam then return end
	local origin = Vector2.new(cam.ViewportSize.X/2, cam.ViewportSize.Y)
	local seen = {}
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character then
			local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if hrp and hum and hum.Health > 0 then
				local skipTeam = false
				if S.TeamCheck then
					local a,b = LP:GetAttribute("TeamID"), plr:GetAttribute("TeamID")
					if a and b and a==b then skipTeam = true end
				end
				if not skipTeam then
					local sp, on = cam:WorldToViewportPoint(hrp.Position)
					if on and sp.Z > 0 then
						seen[plr] = true
						local line = tracerLines[plr]
						if not line then
							line = Instance.new("Frame")
							line.AnchorPoint = Vector2.new(0.5, 0.5)
							line.BorderSizePixel = 0
							line.BackgroundColor3 = Color3.fromRGB(255, 80, 80)
							line.BackgroundTransparency = 0.3
							line.Parent = tracerGui
							tracerLines[plr] = line
						end
						local dest = Vector2.new(sp.X, sp.Y)
						local mid = (origin + dest) / 2
						local dist = (dest - origin).Magnitude
						line.Size = UDim2.fromOffset(math.max(dist,1), 1)
						line.Position = UDim2.fromOffset(mid.X, mid.Y)
						line.Rotation = math.deg(math.atan2(dest.Y-origin.Y, dest.X-origin.X))
					end
				end
			end
		end
	end
	for plr,line in pairs(tracerLines) do
		if not seen[plr] then pcall(function() line:Destroy() end); tracerLines[plr]=nil end
	end
end)
